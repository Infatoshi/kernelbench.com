"""Fused W4A16 Kimi-Linear decode — int4 unpack+dequant fused into GEMV/GEMM."""
from __future__ import annotations

from dataclasses import dataclass, field

import torch
import torch.nn as nn
import torch.nn.functional as F

from w4_kernel import (
    w4a16_batched_gemv,
    w4a16_batched_gemv_varx,
    w4a16_col_gemv_batched,
    w4a16_gemv,
    w4a16_rowcol_gemv_batched,
)

OP_TYPE = "kimi_linear_w4a16_decode"
HARDWARE_REQUIRED = ["RTX_PRO_6000"]
EPS = 1.0e-6
GROUP_SIZE = 128


@dataclass(frozen=True)
class Config:
    hidden: int = 2304
    kda_heads: int = 32
    kda_head_dim: int = 128
    short_conv: int = 4
    mla_heads: int = 32
    kv_lora: int = 512
    qk_nope: int = 128
    qk_rope: int = 64
    v_head: int = 128
    rope_theta: float = 10000.0
    n_experts: int = 64
    n_active: int = 8
    n_shared: int = 1
    moe_inter: int = 1024
    routed_scaling: float = 2.446
    group: int = 128
    pattern: tuple = ("K", "K", "K", "M")
    dtype: torch.dtype = field(default=torch.bfloat16)


def build_config(shape: dict) -> Config:
    return Config(n_experts=int(shape.get("n_experts", 64)))


def _pack_int4(w_q: torch.Tensor) -> torch.Tensor:
    lo = w_q[0::2] & 0xF
    hi = w_q[1::2] & 0xF
    return (lo | (hi << 4)).contiguous()


def _unpack_int4(w_packed: torch.Tensor, K: int) -> torch.Tensor:
    out = torch.empty((K, w_packed.shape[1]), dtype=torch.uint8, device=w_packed.device)
    out[0::2] = w_packed & 0xF
    out[1::2] = (w_packed >> 4) & 0xF
    return out


def quantize(w_io: torch.Tensor, group: int = GROUP_SIZE):
    K, N = w_io.shape
    ng = K // group
    wg = w_io.view(ng, group, N).float()
    wmin = wg.min(dim=1, keepdim=True).values
    wmax = wg.max(dim=1, keepdim=True).values
    scales = (wmax - wmin).clamp_min(1e-8) / 15.0
    zeros = (-wmin / scales).round().clamp(0, 15)
    w_q = ((wg / scales) + zeros).round().clamp(0, 15).to(torch.uint8).view(K, N)
    return _pack_int4(w_q), scales.squeeze(1).to(torch.bfloat16), zeros.squeeze(1).to(torch.bfloat16)


def dequant(w_q: torch.Tensor, scales: torch.Tensor, zeros: torch.Tensor, K: int, group: int) -> torch.Tensor:
    wu = _unpack_int4(w_q, K).to(torch.bfloat16)
    s = scales.repeat_interleave(group, dim=0)
    z = zeros.repeat_interleave(group, dim=0)
    return (wu - z) * s


class QuantLinear(nn.Module):
    def __init__(self, in_f: int, out_f: int, group: int = GROUP_SIZE):
        super().__init__()
        assert in_f % group == 0 and in_f % 2 == 0
        self.in_f, self.out_f, self.group = in_f, out_f, group
        ng = in_f // group
        self.register_buffer("w_q", torch.zeros(in_f // 2, out_f, dtype=torch.uint8))
        self.register_buffer("scales", torch.zeros(ng, out_f, dtype=torch.bfloat16))
        self.register_buffer("zeros", torch.zeros(ng, out_f, dtype=torch.bfloat16))

    def init_random(self, gen: torch.Generator, std: float = 0.02) -> None:
        w = torch.randn(self.in_f, self.out_f, generator=gen) * std
        wq, s, z = quantize(w, self.group)
        self.w_q.copy_(wq)
        self.scales.copy_(s)
        self.zeros.copy_(z)

    def weight_bf(self) -> torch.Tensor:
        return dequant(self.w_q, self.scales, self.zeros, self.in_f, self.group)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        if x.dim() == 1:
            return w4a16_gemv(x.contiguous(), self.w_q, self.scales, self.zeros)
        out = torch.empty(x.shape[0], self.out_f, dtype=torch.bfloat16, device=x.device)
        for i in range(x.shape[0]):
            out[i] = w4a16_gemv(x[i].contiguous(), self.w_q, self.scales, self.zeros)
        return out


class QuantExperts(nn.Module):
    def __init__(self, n: int, in_f: int, out_f: int, group: int = GROUP_SIZE):
        super().__init__()
        self.n, self.in_f, self.out_f, self.group = n, in_f, out_f, group
        ng = in_f // group
        self.register_buffer("w_q", torch.zeros(n, in_f // 2, out_f, dtype=torch.uint8))
        self.register_buffer("scales", torch.zeros(n, ng, out_f, dtype=torch.bfloat16))
        self.register_buffer("zeros", torch.zeros(n, ng, out_f, dtype=torch.bfloat16))

    def init_random(self, gen: torch.Generator, std: float = 0.02) -> None:
        for e in range(self.n):
            w = torch.randn(self.in_f, self.out_f, generator=gen) * std
            wq, s, z = quantize(w, self.group)
            self.w_q[e].copy_(wq)
            self.scales[e].copy_(s)
            self.zeros[e].copy_(z)

    def weight_bf(self, e: int) -> torch.Tensor:
        return dequant(self.w_q[e], self.scales[e], self.zeros[e], self.in_f, self.group)

    def gemv(self, x: torch.Tensor, expert_ids: torch.Tensor) -> torch.Tensor:
        return w4a16_batched_gemv(x.contiguous(), self.w_q, self.scales, self.zeros, expert_ids)

    def gemv_rows(self, x: torch.Tensor, expert_ids: torch.Tensor) -> torch.Tensor:
        return w4a16_batched_gemv_varx(x.contiguous(), self.w_q, self.scales, self.zeros, expert_ids)


def _rmsnorm(x: torch.Tensor, w: torch.Tensor) -> torch.Tensor:
    xf = x.float()
    xf = xf * torch.rsqrt(xf.pow(2).mean(-1, keepdim=True) + EPS)
    return (xf * w.float()).to(x.dtype)


def _rope_cossin(pos: int, dim: int, theta: float, device):
    inv = 1.0 / (theta ** (torch.arange(0, dim, 2, device=device, dtype=torch.float32) / dim))
    ang = pos * inv
    return torch.cos(ang), torch.sin(ang)


def _apply_rope(x: torch.Tensor, cos: torch.Tensor, sin: torch.Tensor) -> torch.Tensor:
    xf = x.float()
    even, odd = xf[..., 0::2], xf[..., 1::2]
    out = torch.empty_like(xf)
    out[..., 0::2] = even * cos - odd * sin
    out[..., 1::2] = odd * cos + even * sin
    return out.to(x.dtype)


class KDA(nn.Module):
    def __init__(self, cfg: Config):
        super().__init__()
        self.cfg = cfg
        H, Dk, d = cfg.kda_heads, cfg.kda_head_dim, cfg.hidden
        self.q_proj = QuantLinear(d, H * Dk, cfg.group)
        self.k_proj = QuantLinear(d, H * Dk, cfg.group)
        self.v_proj = QuantLinear(d, H * Dk, cfg.group)
        self.g_proj = QuantLinear(d, H * Dk, cfg.group)
        self.beta_proj = nn.Linear(d, H, bias=False, dtype=cfg.dtype)
        self.conv_w = nn.Parameter(torch.empty(3, H * Dk, cfg.short_conv, dtype=cfg.dtype))
        self.o_proj = QuantLinear(H * Dk, d, cfg.group)
        self.scale = Dk ** -0.5

    def _short_conv(self, val, prev, idx):
        win = torch.cat([prev, val[None]], dim=0)
        w = self.conv_w[idx].float().transpose(0, 1)
        out = (win.float() * w).sum(0)
        return F.silu(out).to(val.dtype), win[1:]

    def step(self, x, st):
        H, Dk = self.cfg.kda_heads, self.cfg.kda_head_dim
        q = self.q_proj(x)
        k = self.k_proj(x)
        v = self.v_proj(x)
        q, st["cq"] = self._short_conv(q, st["cq"], 0)
        k, st["ck"] = self._short_conv(k, st["ck"], 1)
        v, st["cv"] = self._short_conv(v, st["cv"], 2)
        q = q.view(H, Dk).float() * self.scale
        k = k.view(H, Dk).float()
        v = v.view(H, Dk).float()
        g = (-F.softplus(self.g_proj(x).float())).view(H, Dk)
        beta = torch.sigmoid(self.beta_proj(x).float())
        S = st["S"] * g.exp()[:, :, None]
        pred = (S * k[:, :, None]).sum(1)
        S = S + beta[:, None, None] * k[:, :, None] * (v - pred)[:, None, :]
        o = (S * q[:, :, None]).sum(1)
        st["S"] = S
        return self.o_proj(o.reshape(H * Dk).to(torch.bfloat16))


class MLA(nn.Module):
    def __init__(self, cfg: Config):
        super().__init__()
        self.cfg = cfg
        H, d = cfg.mla_heads, cfg.hidden
        self.q_proj = QuantLinear(d, H * (cfg.qk_nope + cfg.qk_rope), cfg.group)
        self.kv_a = QuantLinear(d, cfg.kv_lora + cfg.qk_rope, cfg.group)
        self.kv_b = QuantLinear(cfg.kv_lora, H * (cfg.qk_nope + cfg.v_head), cfg.group)
        self.o_proj = QuantLinear(H * cfg.v_head, d, cfg.group)
        self.scale = (cfg.qk_nope + cfg.qk_rope) ** -0.5
        self._head_dim = cfg.qk_nope + cfg.v_head

    def step(self, x, st):
        cfg = self.cfg
        H = cfg.mla_heads
        head_dim = self._head_dim
        col_k = torch.arange(H, device=x.device, dtype=torch.int32) * head_dim
        col_v = col_k + cfg.qk_nope
        pos = st["c_kv"].shape[0]
        q = self.q_proj(x).view(H, cfg.qk_nope + cfg.qk_rope)
        q_nope = q[:, : cfg.qk_nope]
        q_rope = q[:, cfg.qk_nope :]
        kv = self.kv_a(x)
        c_kv = kv[: cfg.kv_lora]
        k_rope = kv[cfg.kv_lora :]
        cos, sin = _rope_cossin(pos, cfg.qk_rope, cfg.rope_theta, x.device)
        q_rope = _apply_rope(q_rope, cos, sin)
        k_rope = _apply_rope(k_rope, cos, sin)
        st["c_kv"] = torch.cat([st["c_kv"], c_kv[None]], 0)
        st["k_rope"] = torch.cat([st["k_rope"], k_rope[None]], 0)

        kb = self.kv_b
        absorbed = w4a16_col_gemv_batched(q_nope.contiguous(), kb.w_q, kb.scales, kb.zeros, col_k)
        scores = (
            torch.mm(st["c_kv"], absorbed.T).float()
            + torch.mm(st["k_rope"], q_rope.T).float()
        ) * self.scale
        p = torch.softmax(scores, dim=0).to(x.dtype)
        ctx = torch.mm(p.T, st["c_kv"])
        o = w4a16_rowcol_gemv_batched(ctx.contiguous(), kb.w_q, kb.scales, kb.zeros, col_v)
        return self.o_proj(o.reshape(H * cfg.v_head))


class MoE(nn.Module):
    def __init__(self, cfg: Config):
        super().__init__()
        self.cfg = cfg
        d, m, E = cfg.hidden, cfg.moe_inter, cfg.n_experts
        self.router = nn.Linear(d, E, bias=False, dtype=cfg.dtype)
        self.gate = QuantExperts(E, d, m, cfg.group)
        self.up = QuantExperts(E, d, m, cfg.group)
        self.down = QuantExperts(E, m, d, cfg.group)
        self.s_gate = QuantExperts(cfg.n_shared, d, m, cfg.group)
        self.s_up = QuantExperts(cfg.n_shared, d, m, cfg.group)
        self.s_down = QuantExperts(cfg.n_shared, m, d, cfg.group)

    def _ffn_batched(self, x, experts_g, experts_u, experts_d, idx):
        g = experts_g.gemv(x, idx)
        u = experts_u.gemv(x, idx)
        hh = F.silu(g) * u
        return experts_d.gemv_rows(hh, idx)

    def step(self, x):
        cfg = self.cfg
        probs = torch.softmax(self.router(x).float(), dim=-1)
        w, idx = torch.topk(probs, cfg.n_active)
        w = (w / (w.sum() + 1e-9) * cfg.routed_scaling).to(x.dtype)
        routed = self._ffn_batched(x, self.gate, self.up, self.down, idx)
        out = (w[:, None] * routed).sum(0)
        sidx = torch.arange(cfg.n_shared, device=x.device, dtype=torch.int32)
        shared = self._ffn_batched(x, self.s_gate, self.s_up, self.s_down, sidx)
        out = out + shared.sum(0)
        return out


class Block(nn.Module):
    def __init__(self, cfg: Config, kind: str):
        super().__init__()
        self.kind = kind
        self.attn_norm = nn.Parameter(torch.ones(cfg.hidden, dtype=cfg.dtype))
        self.moe_norm = nn.Parameter(torch.ones(cfg.hidden, dtype=cfg.dtype))
        self.attn = KDA(cfg) if kind == "K" else MLA(cfg)
        self.moe = MoE(cfg)

    def step(self, x, st):
        h = x + self.attn.step(_rmsnorm(x, self.attn_norm), st)
        return h + self.moe.step(_rmsnorm(h, self.moe_norm))


class Model(nn.Module):
    def __init__(self, cfg: Config):
        super().__init__()
        self.cfg = cfg
        self.blocks = nn.ModuleList(Block(cfg, k) for k in cfg.pattern)
        self.reset_parameters()

    def reset_parameters(self):
        g = torch.Generator(device="cpu").manual_seed(1234)
        for mod in self.modules():
            if isinstance(mod, (QuantLinear, QuantExperts)):
                mod.init_random(g)
            elif isinstance(mod, nn.Linear):
                nn.init.normal_(mod.weight, 0.0, 0.02, generator=g)
            elif isinstance(mod, KDA):
                nn.init.normal_(mod.conv_w, 0.0, 0.1, generator=g)

    def step(self, hidden, state):
        for i, blk in enumerate(self.blocks):
            hidden = blk.step(hidden, state[i])
        return hidden, state


def init_state(cfg: Config, context_len: int, seed: int) -> list:
    dev = torch.device("cuda:0")
    g = torch.Generator(device=dev).manual_seed(seed)
    H, Dk = cfg.kda_heads, cfg.kda_head_dim
    C = H * Dk
    state = []
    for kind in cfg.pattern:
        if kind == "K":
            state.append(
                {
                    "S": torch.randn(H, Dk, Dk, device=dev, generator=g) * 0.05,
                    "cq": torch.randn(cfg.short_conv - 1, C, device=dev, generator=g, dtype=cfg.dtype) * 0.1,
                    "ck": torch.randn(cfg.short_conv - 1, C, device=dev, generator=g, dtype=cfg.dtype) * 0.1,
                    "cv": torch.randn(cfg.short_conv - 1, C, device=dev, generator=g, dtype=cfg.dtype) * 0.1,
                }
            )
        else:
            state.append(
                {
                    "c_kv": torch.randn(context_len, cfg.kv_lora, device=dev, generator=g, dtype=cfg.dtype) * 0.1,
                    "k_rope": torch.randn(context_len, cfg.qk_rope, device=dev, generator=g, dtype=cfg.dtype) * 0.1,
                }
            )
    return state


def init_token(cfg: Config, seed: int) -> torch.Tensor:
    dev = torch.device("cuda:0")
    g = torch.Generator(device=dev).manual_seed(seed + 1)
    return torch.randn(cfg.hidden, device=dev, generator=g, dtype=cfg.dtype) * 0.25

"""Fused W4A16 dequant-GEMV / dequant-GEMM — no materialized bf16 weights."""
from __future__ import annotations

import torch
import triton
import triton.language as tl

GROUP = 128
BLOCK_N = 256
BLOCK_M = 64
BLOCK_PAIRS = 64


@triton.jit
def w4a16_gemv_kernel(
    x_ptr,
    wq_ptr,
    scales_ptr,
    zeros_ptr,
    y_ptr,
    K,
    N,
    stride_wq_k,
    stride_wq_n,
    stride_s_g,
    stride_s_n,
    stride_z_g,
    stride_z_n,
    GROUP: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_PAIRS: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_n = pid * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_n = offs_n < N
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)
    ng = K // GROUP

    for g in range(ng):
        k_base = g * GROUP
        kh_base = k_base // 2
        s = tl.load(scales_ptr + g * stride_s_g + offs_n * stride_s_n, mask=mask_n, other=0.0).to(
            tl.float32
        )
        z = tl.load(zeros_ptr + g * stride_z_g + offs_n * stride_z_n, mask=mask_n, other=0.0).to(
            tl.float32
        )
        offs_p = tl.arange(0, BLOCK_PAIRS)
        mask_p = offs_p < GROUP // 2
        x0 = tl.load(x_ptr + k_base + 2 * offs_p, mask=mask_p, other=0.0).to(tl.float32)
        x1 = tl.load(x_ptr + k_base + 2 * offs_p + 1, mask=mask_p, other=0.0).to(tl.float32)
        packed = tl.load(
            wq_ptr + (kh_base + offs_p)[:, None] * stride_wq_k + offs_n[None, :] * stride_wq_n,
            mask=mask_p[:, None] & mask_n[None, :],
            other=0,
        ).to(tl.int32)
        w0 = (packed & 0xF).to(tl.float32)
        w1 = ((packed >> 4) & 0xF).to(tl.float32)
        acc += tl.sum(x0[:, None] * (w0 - z[None, :]) * s[None, :], axis=0)
        acc += tl.sum(x1[:, None] * (w1 - z[None, :]) * s[None, :], axis=0)

    tl.store(y_ptr + offs_n, acc.to(tl.bfloat16), mask=mask_n)


@triton.jit
def w4a16_col_gemv_kernel(
    x_ptr,
    wq_ptr,
    scales_ptr,
    zeros_ptr,
    y_ptr,
    col_start,
    K,
    Kslice,
    stride_wq_k,
    stride_wq_n,
    stride_s_g,
    stride_s_n,
    stride_z_g,
    stride_z_n,
    GROUP: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """y[c] = W[c, col_start:col_start+Kslice] @ x  with W shape (K, N_full)."""
    pid = tl.program_id(0)
    offs_c = pid * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_c = offs_c < K
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)
    kh = offs_c // 2
    is_even = (offs_c % 2) == 0
    g_row = offs_c // GROUP

    for d in range(0, Kslice, 2):
        col0 = col_start + d
        col1 = col_start + d + 1
        x0 = tl.load(x_ptr + d).to(tl.float32)
        x1 = tl.load(x_ptr + d + 1).to(tl.float32)

        pk0 = tl.load(
            wq_ptr + kh * stride_wq_k + col0 * stride_wq_n, mask=mask_c, other=0
        ).to(tl.int32)
        pk1 = tl.load(
            wq_ptr + kh * stride_wq_k + col1 * stride_wq_n, mask=mask_c, other=0
        ).to(tl.int32)
        w0 = tl.where(is_even, pk0 & 0xF, (pk0 >> 4) & 0xF).to(tl.float32)
        w1 = tl.where(is_even, pk1 & 0xF, (pk1 >> 4) & 0xF).to(tl.float32)

        s0 = tl.load(
            scales_ptr + g_row * stride_s_g + col0 * stride_s_n, mask=mask_c, other=0.0
        ).to(tl.float32)
        z0 = tl.load(
            zeros_ptr + g_row * stride_z_g + col0 * stride_z_n, mask=mask_c, other=0.0
        ).to(tl.float32)
        s1 = tl.load(
            scales_ptr + g_row * stride_s_g + col1 * stride_s_n, mask=mask_c, other=0.0
        ).to(tl.float32)
        z1 = tl.load(
            zeros_ptr + g_row * stride_z_g + col1 * stride_z_n, mask=mask_c, other=0.0
        ).to(tl.float32)

        acc += x0 * (w0 - z0) * s0
        acc += x1 * (w1 - z1) * s1

    tl.store(y_ptr + offs_c, acc.to(tl.bfloat16), mask=mask_c)


@triton.jit
def w4a16_col_gemv_batched_kernel(
    x_ptr,
    wq_ptr,
    scales_ptr,
    zeros_ptr,
    y_ptr,
    col_starts_ptr,
    K,
    Kslice,
    stride_x_h,
    stride_x_k,
    stride_y_h,
    stride_y_k,
    stride_wq_k,
    stride_wq_n,
    stride_s_g,
    stride_s_n,
    stride_z_g,
    stride_z_n,
    GROUP: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_h = tl.program_id(0)
    pid_c = tl.program_id(1)
    col_start = tl.load(col_starts_ptr + pid_h).to(tl.int32)
    x_base = x_ptr + pid_h * stride_x_h
    y_base = y_ptr + pid_h * stride_y_h

    offs_c = pid_c * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_c = offs_c < K
    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)
    kh = offs_c // 2
    is_even = (offs_c % 2) == 0
    g_row = offs_c // GROUP

    for d in range(0, Kslice, 2):
        col0 = col_start + d
        col1 = col_start + d + 1
        x0 = tl.load(x_base + d * stride_x_k).to(tl.float32)
        x1 = tl.load(x_base + (d + 1) * stride_x_k).to(tl.float32)

        pk0 = tl.load(
            wq_ptr + kh * stride_wq_k + col0 * stride_wq_n, mask=mask_c, other=0
        ).to(tl.int32)
        pk1 = tl.load(
            wq_ptr + kh * stride_wq_k + col1 * stride_wq_n, mask=mask_c, other=0
        ).to(tl.int32)
        w0 = tl.where(is_even, pk0 & 0xF, (pk0 >> 4) & 0xF).to(tl.float32)
        w1 = tl.where(is_even, pk1 & 0xF, (pk1 >> 4) & 0xF).to(tl.float32)

        s0 = tl.load(
            scales_ptr + g_row * stride_s_g + col0 * stride_s_n, mask=mask_c, other=0.0
        ).to(tl.float32)
        z0 = tl.load(
            zeros_ptr + g_row * stride_z_g + col0 * stride_z_n, mask=mask_c, other=0.0
        ).to(tl.float32)
        s1 = tl.load(
            scales_ptr + g_row * stride_s_g + col1 * stride_s_n, mask=mask_c, other=0.0
        ).to(tl.float32)
        z1 = tl.load(
            zeros_ptr + g_row * stride_z_g + col1 * stride_z_n, mask=mask_c, other=0.0
        ).to(tl.float32)

        acc += x0 * (w0 - z0) * s0
        acc += x1 * (w1 - z1) * s1

    tl.store(y_base + offs_c * stride_y_k, acc.to(tl.bfloat16), mask=mask_c)


@triton.jit
def w4a16_rowcol_gemv_batched_kernel(
    x_ptr,
    wq_ptr,
    scales_ptr,
    zeros_ptr,
    y_ptr,
    col_starts_ptr,
    K,
    Kslice,
    stride_x_h,
    stride_x_k,
    stride_y_h,
    stride_y_d,
    stride_wq_k,
    stride_wq_n,
    stride_s_g,
    stride_s_n,
    stride_z_g,
    stride_z_n,
    GROUP: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """y[h,d] = sum_c x[h,c] * W[c, col_start+d]  (output length Kslice)."""
    pid_h = tl.program_id(0)
    pid_d = tl.program_id(1)
    col_start = tl.load(col_starts_ptr + pid_h).to(tl.int32)
    x_base = x_ptr + pid_h * stride_x_h
    y_base = y_ptr + pid_h * stride_y_h

    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)
    mask_d = offs_d < Kslice
    acc = tl.zeros((BLOCK_D,), dtype=tl.float32)

    for c in range(K):
        xh = tl.load(x_base + c * stride_x_k).to(tl.float32)
        kh = c // 2
        is_even = (c % 2) == 0
        g_row = c // GROUP

        col = col_start + offs_d
        pk = tl.load(
            wq_ptr + kh * stride_wq_k + col * stride_wq_n, mask=mask_d, other=0
        ).to(tl.int32)
        w = tl.where(is_even, pk & 0xF, pk >> 4).to(tl.float32)
        s = tl.load(
            scales_ptr + g_row * stride_s_g + col * stride_s_n, mask=mask_d, other=0.0
        ).to(tl.float32)
        z = tl.load(
            zeros_ptr + g_row * stride_z_g + col * stride_z_n, mask=mask_d, other=0.0
        ).to(tl.float32)
        acc += xh * (w - z) * s

    tl.store(y_base + offs_d * stride_y_d, acc.to(tl.bfloat16), mask=mask_d)


@triton.jit
def w4a16_batched_gemv_kernel(
    x_ptr,
    wq_ptr,
    scales_ptr,
    zeros_ptr,
    y_ptr,
    expert_ids_ptr,
    K,
    N,
    stride_wq_e,
    stride_wq_k,
    stride_wq_n,
    stride_se_e,
    stride_se_g,
    stride_se_n,
    stride_ze_e,
    stride_ze_g,
    stride_ze_n,
    stride_y_e,
    stride_y_n,
    GROUP: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_PAIRS: tl.constexpr,
):
    pid_e = tl.program_id(0)
    pid_n = tl.program_id(1)
    eid = tl.load(expert_ids_ptr + pid_e).to(tl.int32)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_n = offs_n < N

    wq_base = wq_ptr + eid * stride_wq_e
    s_base = scales_ptr + eid * stride_se_e
    z_base = zeros_ptr + eid * stride_ze_e
    y_base = y_ptr + pid_e * stride_y_e

    acc = tl.zeros((BLOCK_N,), dtype=tl.float32)
    ng = K // GROUP

    for g in range(ng):
        k_base = g * GROUP
        kh_base = k_base // 2
        s = tl.load(
            s_base + g * stride_se_g + offs_n * stride_se_n, mask=mask_n, other=0.0
        ).to(tl.float32)
        z = tl.load(
            z_base + g * stride_ze_g + offs_n * stride_ze_n, mask=mask_n, other=0.0
        ).to(tl.float32)
        offs_p = tl.arange(0, BLOCK_PAIRS)
        mask_p = offs_p < GROUP // 2
        x0 = tl.load(x_ptr + k_base + 2 * offs_p, mask=mask_p, other=0.0).to(tl.float32)
        x1 = tl.load(x_ptr + k_base + 2 * offs_p + 1, mask=mask_p, other=0.0).to(tl.float32)
        packed = tl.load(
            wq_base + (kh_base + offs_p)[:, None] * stride_wq_k + offs_n[None, :] * stride_wq_n,
            mask=mask_p[:, None] & mask_n[None, :],
            other=0,
        ).to(tl.int32)
        w0 = (packed & 0xF).to(tl.float32)
        w1 = ((packed >> 4) & 0xF).to(tl.float32)
        acc += tl.sum(x0[:, None] * (w0 - z[None, :]) * s[None, :], axis=0)
        acc += tl.sum(x1[:, None] * (w1 - z[None, :]) * s[None, :], axis=0)

    tl.store(y_base + offs_n * stride_y_n, acc.to(tl.bfloat16), mask=mask_n)

def _launch_gemv(x, w_q, scales, zeros, y):
    K = w_q.shape[0] * 2
    N = w_q.shape[1]
    grid = (triton.cdiv(N, BLOCK_N),)
    w4a16_gemv_kernel[grid](
        x,
        w_q,
        scales,
        zeros,
        y,
        K,
        N,
        w_q.stride(0),
        w_q.stride(1),
        scales.stride(0),
        scales.stride(1),
        zeros.stride(0),
        zeros.stride(1),
        GROUP=GROUP,
        BLOCK_N=BLOCK_N,
        BLOCK_PAIRS=BLOCK_PAIRS,
        num_warps=8,
    )


def w4a16_gemv(x: torch.Tensor, w_q: torch.Tensor, scales: torch.Tensor, zeros: torch.Tensor) -> torch.Tensor:
    y = torch.empty(w_q.shape[1], dtype=torch.bfloat16, device=x.device)
    _launch_gemv(x.contiguous(), w_q, scales, zeros, y)
    return y


def w4a16_gemm(x: torch.Tensor, w_q: torch.Tensor, scales: torch.Tensor, zeros: torch.Tensor) -> torch.Tensor:
    """GEMM via per-row GEMV (fallback for non-MLA paths)."""
    M = x.shape[0]
    N = w_q.shape[1]
    y = torch.empty(M, N, dtype=torch.bfloat16, device=x.device)
    for i in range(M):
        _launch_gemv(x[i].contiguous(), w_q, scales, zeros, y[i])
    return y

        x.stride(1),

        y.stride(1),
        w_q.stride(0),
        w_q.stride(1),
        scales.stride(0),
        scales.stride(1),
        zeros.stride(0),
        zeros.stride(1),
        GROUP=GROUP,
        BLOCK_N=BLOCK_N,
        num_warps=8,
    )
    return y
