"""Vectorized grid foraging + 3-layer MinGRU policy, CUDA kernels.

solution.py exposes:
  class Model             (state_dict-compatible with reference.Model)
  policy_forward(model, obs, state) -> (logits, new_state, value)
  env_step(agent, food, actions, rng_state) -> (agent, food, reward, rng_state)
  run(num_envs, horizon, seed, model=None) -> dict

The graded path is run(): a cooperative megakernel that fuses the full
horizon of env steps + greedy policy. policy_forward / env_step exist for the
correctness checks and use a small per-env policy kernel / reference math.
"""

_CUDA_SRC = r"""
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <cuda_runtime.h>
#include <cooperative_groups.h>
namespace cg = cooperative_groups;

#define H 256
#define L 3
#define G 768
#define NACT 4
#define BOARD 11
#define LCG_MUL 6364136223846793005ULL
#define LCG_MASK 0x7FFFFFFFFFFFFFFFULL

__device__ __forceinline__ long long lcg_step(long long r) {
  unsigned long long v = (unsigned long long)r * LCG_MUL + 1ULL;
  return (long long)(v & LCG_MASK);
}

__device__ __forceinline__ float sigf(float x) {
  return 1.0f / (1.0f + expf(-x));
}

// One full policy step: obs[4] -> h (256) through 3 MinGRU layers.
// st is the persistent (L*H) state, in/out. h is scratch, in/out.
// zh/zg/zp are scratch buffers of size H each. lg[4] logits, *val value.
__device__ __noinline__ void policy_step_device(
    const float* __restrict__ obs,
    float* st, float* h, float* zh, float* zg, float* zp,
    float* lg, float* val,
    const float* __restrict__ w_enc, const float* __restrict__ b_enc,
    const float* __restrict__ w_gru,
    const float* __restrict__ w_a, const float* __restrict__ b_a,
    const float* __restrict__ w_v, const float* __restrict__ b_v)
{
  // Linear(4 -> 256)
  for (int n = 0; n < H; n++) {
    float a = 0.0f;
    for (int j = 0; j < 4; j++) a += obs[j] * w_enc[n * 4 + j];
    h[n] = a + b_enc[n];
  }
  // MinGRU x3
  for (int l = 0; l < L; l++) {
    const float* wl = w_gru + (size_t)l * G * H;
    for (int n = 0; n < H; n++) {
      const float* p0 = wl + n * H;
      const float* p1 = wl + H * H + n * H;
      const float* p2 = wl + 2 * H * H + n * H;
      float a0 = 0.0f, a1 = 0.0f, a2 = 0.0f;
      for (int j = 0; j < H; j++) {
        float hj = h[j];
        a0 += hj * p0[j];
        a1 += hj * p1[j];
        a2 += hj * p2[j];
      }
      zh[n] = a0; zg[n] = a1; zp[n] = a2;
    }
    for (int n = 0; n < H; n++) {
      float sg = sigf(zg[n]);
      float th = tanhf(zh[n]);
      float out = st[l * H + n] + sg * (th - st[l * H + n]);
      float p = sigf(zp[n]);
      st[l * H + n] = out;
      h[n] = p * out + (1.0f - p) * h[n];
    }
  }
  // action logits (256 -> 4) and value (256 -> 1)
  for (int a = 0; a < NACT; a++) {
    float acc = 0.0f;
    for (int j = 0; j < H; j++) acc += h[j] * w_a[a * H + j];
    lg[a] = acc + b_a[a];
  }
  float vv = b_v[0];
  for (int j = 0; j < H; j++) vv += h[j] * w_v[j];
  *val = vv;
}

// Standalone policy forward (used by policy_forward for the correctness check).
__global__ void policy_kernel(
    const float* __restrict__ obs, const float* __restrict__ state,
    const float* __restrict__ w_enc, const float* __restrict__ b_enc,
    const float* __restrict__ w_gru, const float* __restrict__ w_a, const float* __restrict__ b_a,
    const float* __restrict__ w_v, const float* __restrict__ b_v,
    float* __restrict__ logits, float* __restrict__ new_state, float* __restrict__ value,
    int N)
{
  int i = blockIdx.x * blockDim.x + threadIdx.x;
  if (i >= N) return;
  float st[L * H];
  float h[H];
  float zh[H], zg[H], zp[H];
  float lg[NACT];
  float val;
  for (int l = 0; l < L; l++)
    for (int n = 0; n < H; n++)
      st[l * H + n] = state[(i * L + l) * H + n];
  policy_step_device(&obs[i * 4], st, h, zh, zg, zp, lg, &val,
                     w_enc, b_enc, w_gru, w_a, b_a, w_v, b_v);
  for (int a = 0; a < NACT; a++) logits[i * NACT + a] = lg[a];
  for (int l = 0; l < L; l++)
    for (int n = 0; n < H; n++)
      new_state[(i * L + l) * H + n] = st[l * H + n];
  value[i] = val;
}

// Fused rollout megakernel: one thread per env, horizon loop inside, one
// grid-wide sync per step to resolve the "did any env hit" LCG dependency.
__global__ void rollout_kernel(
    const float* __restrict__ w_enc, const float* __restrict__ b_enc,
    const float* __restrict__ w_gru, const float* __restrict__ w_a, const float* __restrict__ b_a,
    const float* __restrict__ w_v, const float* __restrict__ b_v,
    const float* __restrict__ agent0, const float* __restrict__ food0,
    const long long* __restrict__ rng0,
    float* __restrict__ rewards, long long* __restrict__ positions,
    float* __restrict__ last_logits,
    int N, int horizon, unsigned long long* __restrict__ flag)
{
  int i = blockIdx.x * blockDim.x + threadIdx.x;
  bool active = (i < N);

  float ax = 0.0f, ay = 0.0f, fx = 0.0f, fy = 0.0f;
  long long rng = 0;
  float st[L * H];
  float h[H];
  float zh[H], zg[H], zp[H];
  float lg[NACT];
  float val;
  float reward = 0.0f;
  int hit = 0;

  if (active) {
    ax = agent0[i * 2]; ay = agent0[i * 2 + 1];
    fx = food0[i * 2]; fy = food0[i * 2 + 1];
    rng = rng0[i];
    for (int k = 0; k < L * H; k++) st[k] = 0.0f;
  }

  for (int t = 0; t < horizon; t++) {
    if (active) {
      float obs[4];
      obs[0] = (fx - ax) / (float)BOARD;
      obs[1] = (fy - ay) / (float)BOARD;
      obs[2] = ax / 10.0f;
      obs[3] = ay / 10.0f;

      policy_step_device(obs, st, h, zh, zg, zp, lg, &val,
                         w_enc, b_enc, w_gru, w_a, b_a, w_v, b_v);

      // greedy argmax (first index wins on ties, matching torch.argmax)
      int action = 0;
      float best = lg[0];
      if (lg[1] > best) { best = lg[1]; action = 1; }
      if (lg[2] > best) { best = lg[2]; action = 2; }
      if (lg[3] > best) { best = lg[3]; action = 3; }

      int dx = 0, dy = 0;
      if (action == 0) dy = -1;
      else if (action == 1) dy = 1;
      else if (action == 2) dx = -1;
      else dx = 1;

      int nx = (int)ax + dx; if (nx < 0) nx = 0; if (nx > BOARD - 1) nx = BOARD - 1;
      int ny = (int)ay + dy; if (ny < 0) ny = 0; if (ny > BOARD - 1) ny = BOARD - 1;
      ax = (float)nx; ay = (float)ny;

      hit = (nx == (int)fx && ny == (int)fy) ? 1 : 0;
      if (hit) { reward += 1.0f; atomicOr(flag, 1); }

      if (t == horizon - 1) {
        last_logits[i * NACT + 0] = lg[0];
        last_logits[i * NACT + 1] = lg[1];
        last_logits[i * NACT + 2] = lg[2];
        last_logits[i * NACT + 3] = lg[3];
      }
    }

    cg::this_grid().sync();
    if (active && (atomicOr(flag, 0) != 0)) {
      rng = lcg_step(rng);
      int nx2 = (int)(rng % BOARD);
      rng = lcg_step(rng);
      int ny2 = (int)(rng % BOARD);
      if (hit) { fx = (float)nx2; fy = (float)ny2; }
    }
    cg::this_grid().sync();
    if (blockIdx.x == 0 && threadIdx.x == 0) atomicExch(flag, 0);
    cg::this_grid().sync();
  }

  if (active) {
    rewards[i] = reward;
    positions[i * 2] = (long long)ax;
    positions[i * 2 + 1] = (long long)ay;
  }
}

// ---------------------------------------------------------------------------
// V2: register-tiled GEMM megakernel. One persistent cooperative block per SM;
// each block owns a strip of envs and processes them in TILE-wide chunks.
// h lives in shared memory, gates are computed with a 4x8 register-tile SGEMM
// (weights read from L2, not staged), and the global "any hit" LCG dependency
// is resolved with three grid-wide syncs per step.
// ---------------------------------------------------------------------------

#define TILE 32
#define NTHREADS 256
#define TM 4
#define TN 8
#define BK 8

__device__ __forceinline__ void cp_async16(float* smem_dst, const float* gmem_src) {
  unsigned s = (unsigned)__cvta_generic_to_shared(smem_dst);
  asm volatile("cp.async.cg.shared.global [%0], [%1], 16;\n" :: "r"(s), "l"(gmem_src));
}
__device__ __forceinline__ void cp_async_commit() {
  asm volatile("cp.async.commit_group;\n");
}
template <int N> __device__ __forceinline__ void cp_async_wait() {
  asm volatile("cp.async.wait_group %0;\n" :: "n"(N));
}

__device__ __forceinline__ int col_of(int nt, int j) {
  // Column-split mapping so the 32 lanes of a warp read fully-coalesced float4s:
  // acc[i][0..3] -> columns nt*4..nt*4+3 ; acc[i][4..7] -> columns 128+nt*4..+3.
  return (j < 4) ? (nt * 4 + j) : (128 + nt * 4 + (j - 4));
}

// Logits = h @ w_a^T + b_a, one (env, action) output per thread. 256 threads
// cover TILE*4 outputs (64*4 or 32*4); each does a full K=256 dot so no
// cross-thread reduction is needed. Replaces the serial per-env matvec that
// previously ran on only `nenv` threads.
__device__ __forceinline__ void logits_parallel(const float* __restrict__ h_s,
                                                const float* __restrict__ w_a, const float* __restrict__ b_a,
                                                float* __restrict__ lg_s, int tid, int nrow) {
  if (tid >= nrow * 4) return;   // nrow = valid env rows in this tile (<= TILE)
  int e = tid >> 2;
  int a = tid & 3;
  float acc = b_a[a];
  const float* wa = w_a + a * H;
  const float* hs = h_s + e * H;
  #pragma unroll 8
  for (int n = 0; n < H; n++) acc += hs[n] * wa[n];
  lg_s[tid] = acc;
}

__device__ __forceinline__ void gemm_gate_acc(const float* __restrict__ h_s,
                                              const float* __restrict__ W,
                                              float* __restrict__ b_s,
                                              float acc[TM][TN],
                                              int mt, int nt, int tid) {
  // cp.async double-buffer: prefetch the next K-tile's weights into shared
  // while computing the current tile, hiding global-load latency.
  float* buf0 = b_s;
  float* buf1 = b_s + BK * H;
  const int nkt = H / BK;              // 32
  const int nld = BK * H / NTHREADS / 4; // float4 loads per thread per tile
  // prologue: async load tile 0 -> buf0
  {
    const float4* Wf = (const float4*)(W);
    float4* dst = (float4*)buf0;
    #pragma unroll
    for (int u = 0; u < nld; u++)
      cp_async16((float*)(dst + u * NTHREADS + tid), (const float*)(Wf + u * NTHREADS + tid));
    cp_async_commit();
  }
  for (int kk = 0; kk < nkt; kk++) {
    float* cur = (kk & 1) ? buf1 : buf0;
    float* nxt = (kk & 1) ? buf0 : buf1;
    if (kk + 1 < nkt) {
      const float4* Wf = (const float4*)(W + (kk + 1) * BK * H);
      float4* dst = (float4*)nxt;
      #pragma unroll
      for (int u = 0; u < nld; u++)
        cp_async16((float*)(dst + u * NTHREADS + tid), (const float*)(Wf + u * NTHREADS + tid));
      cp_async_commit();
    }
    cp_async_wait<1>();
    __syncthreads();
    #pragma unroll
    for (int k = 0; k < BK; k++) {
      float hs[TM];
      #pragma unroll
      for (int i = 0; i < TM; i++) hs[i] = h_s[(mt * TM + i) * H + kk * BK + k];
      const float4* bp0 = (const float4*)(cur + k * H + nt * 4);
      const float4* bp1 = (const float4*)(cur + k * H + 128 + nt * 4);
      float4 b0 = bp0[0];
      float4 b1 = bp1[0];
      #pragma unroll
      for (int i = 0; i < TM; i++) {
        float h = hs[i];
        acc[i][0] += h * b0.x; acc[i][1] += h * b0.y; acc[i][2] += h * b0.z; acc[i][3] += h * b0.w;
        acc[i][4] += h * b1.x; acc[i][5] += h * b1.y; acc[i][6] += h * b1.z; acc[i][7] += h * b1.w;
      }
    }
    __syncthreads();
  }
}

__global__ void __launch_bounds__(NTHREADS, 2) rollout_kernel_v2(
    const float* __restrict__ w_enc, const float* __restrict__ b_enc,
    const float* __restrict__ w_gru, const float* __restrict__ w_a, const float* __restrict__ b_a,
    float* __restrict__ agent0, float* __restrict__ food0, long long* __restrict__ rng0,
    float* __restrict__ state_buf, int* __restrict__ hit_arr,
    float* __restrict__ rewards, long long* __restrict__ positions,
    float* __restrict__ last_logits,
    int N, int horizon, unsigned long long* __restrict__ flag)
{
  extern __shared__ float smem[];
  float* h_s = smem;                       // TILE*H
  float* obs_s = h_s + TILE * H;           // TILE*4
  float* b_s = obs_s + TILE * 4;           // 2*BK*H (double-buffered weight tile)

  const int tid = threadIdx.x;
  const int mt = tid >> 5;   // 0..7  (M-tile index)
  const int nt = tid & 31;   // 0..31 (N-tile index)

  int num_tiles = (N + TILE - 1) / TILE;

  for (int t = 0; t < horizon; t++) {
    for (int tile = blockIdx.x; tile < num_tiles; tile += gridDim.x) {
      int base = tile * TILE;
      int nenv = N - base; if (nenv > TILE) nenv = TILE;

      // obs (per env, 4 floats)
      if (tid < nenv) {
        int env = base + tid;
        float ax = agent0[env * 2], ay = agent0[env * 2 + 1];
        float fx = food0[env * 2], fy = food0[env * 2 + 1];
        obs_s[tid * 4 + 0] = (fx - ax) * (1.0f / (float)BOARD);
        obs_s[tid * 4 + 1] = (fy - ay) * (1.0f / (float)BOARD);
        obs_s[tid * 4 + 2] = ax * 0.1f;
        obs_s[tid * 4 + 3] = ay * 0.1f;
      }
      __syncthreads();

      // enc: h_s[e][n] = obs[e] . w_enc[n] + b_enc[n]
      for (int idx = tid; idx < nenv * H; idx += NTHREADS) {
        int e = idx / H, n = idx % H;
        const float* o = obs_s + e * 4;
        h_s[e * H + n] = o[0] * w_enc[n * 4 + 0] + o[1] * w_enc[n * 4 + 1]
                       + o[2] * w_enc[n * 4 + 2] + o[3] * w_enc[n * 4 + 3] + b_enc[n];
      }
      __syncthreads();

      for (int l = 0; l < L; l++) {
        const float* wl = w_gru + (size_t)l * G * H;
        float azh[TM][TN], azg[TM][TN];
        #pragma unroll
        for (int i = 0; i < TM; i++)
          #pragma unroll
          for (int j = 0; j < TN; j++) { azh[i][j] = 0.0f; azg[i][j] = 0.0f; }
        gemm_gate_acc(h_s, wl, b_s, azh, mt, nt, tid);            // zh
        gemm_gate_acc(h_s, wl + H * H, b_s, azg, mt, nt, tid);    // zg
        // out = st + sig(zg) * (tanh(zh) - st)  (in registers, no shared gate buffer)
        #pragma unroll
        for (int i = 0; i < TM; i++)
          #pragma unroll
          for (int j = 0; j < TN; j++) {
            int e = mt * TM + i, n = col_of(nt, j);
            if (e < nenv) {
              int env = base + e;
              float st = state_buf[(env * L + l) * H + n];
              float sg = sigf(azg[i][j]);
              float th = tanhf(azh[i][j]);
              float out = st + sg * (th - st);
              state_buf[(env * L + l) * H + n] = out;
              azh[i][j] = out;   // stash out for combine2
            }
          }
        #pragma unroll
        for (int i = 0; i < TM; i++)
          #pragma unroll
          for (int j = 0; j < TN; j++) azg[i][j] = 0.0f;
        gemm_gate_acc(h_s, wl + 2 * H * H, b_s, azg, mt, nt, tid);  // zp
        __syncthreads();
        // h_new = sig(zp) * out + (1 - sig(zp)) * h_old
        #pragma unroll
        for (int i = 0; i < TM; i++)
          #pragma unroll
          for (int j = 0; j < TN; j++) {
            int e = mt * TM + i, n = col_of(nt, j);
            if (e < nenv) {
              float p = sigf(azg[i][j]);
              float out = azh[i][j];
              float ho = h_s[e * H + n];
              h_s[e * H + n] = p * out + (1.0f - p) * ho;
            }
          }
        __syncthreads();
      }

      // logits + greedy env step
      logits_parallel(h_s, w_a, b_a, obs_s, tid, nenv);
      __syncthreads();
      if (tid < nenv) {
        int env = base + tid;
        float lg0 = obs_s[tid * 4 + 0], lg1 = obs_s[tid * 4 + 1];
        float lg2 = obs_s[tid * 4 + 2], lg3 = obs_s[tid * 4 + 3];
        int action = 0; float best = lg0;
        if (lg1 > best) { best = lg1; action = 1; }
        if (lg2 > best) { best = lg2; action = 2; }
        if (lg3 > best) { best = lg3; action = 3; }
        int dx = 0, dy = 0;
        if (action == 0) dy = -1;
        else if (action == 1) dy = 1;
        else if (action == 2) dx = -1;
        else dx = 1;
        float ax = agent0[env * 2], ay = agent0[env * 2 + 1];
        int nx = (int)ax + dx; if (nx < 0) nx = 0; if (nx > BOARD - 1) nx = BOARD - 1;
        int ny = (int)ay + dy; if (ny < 0) ny = 0; if (ny > BOARD - 1) ny = BOARD - 1;
        agent0[env * 2] = (float)nx; agent0[env * 2 + 1] = (float)ny;
        int hit = (nx == (int)food0[env * 2] && ny == (int)food0[env * 2 + 1]) ? 1 : 0;
        hit_arr[env] = hit;
        if (hit) { rewards[env] += 1.0f; atomicOr(&flag[t >> 6], 1ULL << (t & 63)); }
        if (t == horizon - 1) {
          last_logits[env * 4 + 0] = lg0;
          last_logits[env * 4 + 1] = lg1;
          last_logits[env * 4 + 2] = lg2;
          last_logits[env * 4 + 3] = lg3;
          positions[env * 2] = (long long)nx;
          positions[env * 2 + 1] = (long long)ny;
        }
      }
    }

    cg::this_grid().sync();
    int any_hit = (int)((atomicOr(&flag[t >> 6], 0) >> (t & 63)) & 1ULL);
    if (any_hit != 0) {
      for (int tile = blockIdx.x; tile < num_tiles; tile += gridDim.x) {
        int base = tile * TILE;
        int nenv = N - base; if (nenv > TILE) nenv = TILE;
        for (int e = tid; e < nenv; e += NTHREADS) {
          int env = base + e;
          long long r = rng0[env];
          r = lcg_step(r);
          int nx2 = (int)(r % BOARD);
          r = lcg_step(r);
          int ny2 = (int)(r % BOARD);
          rng0[env] = r;
          if (hit_arr[env]) { food0[env * 2] = (float)nx2; food0[env * 2 + 1] = (float)ny2; }
        }
      }
    }
  }
}

// ---------------------------------------------------------------------------
// V3: TILE=64, TM=8, TN=8 (64 accumulators/thread/gate). Fuses zh+zg into one
// GEMM (h read once, both gate accumulators live) and halves B-read redundancy
// (8 -> 4 warps per column). 1 block/SM (~97KB shared) with 256 regs/thread.
// ---------------------------------------------------------------------------
#define V3_TILE 64
#define V3_TM 8
#define V3_TN 8

__device__ __forceinline__ void gemm_gate8(const float* __restrict__ h_s,
                                           const float* __restrict__ W,
                                           float* __restrict__ b_s,
                                           float acc[V3_TM][V3_TN],
                                           int mt, int nt, int tid) {
  float* buf0 = b_s;
  float* buf1 = b_s + BK * H;
  const int nkt = H / BK;
  const int nld = BK * H / NTHREADS / 4;
  {
    const float4* Wf = (const float4*)(W);
    float4* dst = (float4*)buf0;
    #pragma unroll
    for (int u = 0; u < nld; u++)
      cp_async16((float*)(dst + u * NTHREADS + tid), (const float*)(Wf + u * NTHREADS + tid));
    cp_async_commit();
  }
  for (int kk = 0; kk < nkt; kk++) {
    float* cur = (kk & 1) ? buf1 : buf0;
    float* nxt = (kk & 1) ? buf0 : buf1;
    if (kk + 1 < nkt) {
      const float4* Wf = (const float4*)(W + (kk + 1) * BK * H);
      float4* dst = (float4*)nxt;
      #pragma unroll
      for (int u = 0; u < nld; u++)
        cp_async16((float*)(dst + u * NTHREADS + tid), (const float*)(Wf + u * NTHREADS + tid));
      cp_async_commit();
    }
    cp_async_wait<1>();
    __syncthreads();
    #pragma unroll
    for (int k = 0; k < BK; k++) {
      float hs[V3_TM];
      #pragma unroll
      for (int i = 0; i < V3_TM; i++) hs[i] = h_s[(mt * V3_TM + i) * H + kk * BK + k];
      const float4* bp0 = (const float4*)(cur + k * H + nt * 4);
      const float4* bp1 = (const float4*)(cur + k * H + 128 + nt * 4);
      float4 b0 = bp0[0];
      float4 b1 = bp1[0];
      #pragma unroll
      for (int i = 0; i < V3_TM; i++) {
        float h = hs[i];
        acc[i][0] += h * b0.x; acc[i][1] += h * b0.y; acc[i][2] += h * b0.z; acc[i][3] += h * b0.w;
        acc[i][4] += h * b1.x; acc[i][5] += h * b1.y; acc[i][6] += h * b1.z; acc[i][7] += h * b1.w;
      }
    }
    __syncthreads();
  }
}

__device__ __forceinline__ void gemm_fused2_8(const float* __restrict__ h_s,
                                              const float* __restrict__ Wzh,
                                              const float* __restrict__ Wzg,
                                              float* __restrict__ b_s,
                                              float azh[V3_TM][V3_TN], float azg[V3_TM][V3_TN],
                                              int mt, int nt, int tid) {
  float* buf0 = b_s;
  float* buf1 = b_s + 2 * BK * H;
  const int nkt = H / BK;
  const int nq = BK * H / 4;        // float4 per gate tile
  const int nld = nq / NTHREADS;    // per thread per gate
  {
    const float4* Wf0 = (const float4*)(Wzh);
    const float4* Wf1 = (const float4*)(Wzg);
    float4* dst = (float4*)buf0;
    #pragma unroll
    for (int u = 0; u < nld; u++) {
      cp_async16((float*)(dst + u * NTHREADS + tid), (const float*)(Wf0 + u * NTHREADS + tid));
      cp_async16((float*)(dst + nq + u * NTHREADS + tid), (const float*)(Wf1 + u * NTHREADS + tid));
    }
    cp_async_commit();
  }
  for (int kk = 0; kk < nkt; kk++) {
    float* cur = (kk & 1) ? buf1 : buf0;
    float* nxt = (kk & 1) ? buf0 : buf1;
    if (kk + 1 < nkt) {
      const float4* Wf0 = (const float4*)(Wzh + (kk + 1) * BK * H);
      const float4* Wf1 = (const float4*)(Wzg + (kk + 1) * BK * H);
      float4* dst = (float4*)nxt;
      #pragma unroll
      for (int u = 0; u < nld; u++) {
        cp_async16((float*)(dst + u * NTHREADS + tid), (const float*)(Wf0 + u * NTHREADS + tid));
        cp_async16((float*)(dst + nq + u * NTHREADS + tid), (const float*)(Wf1 + u * NTHREADS + tid));
      }
      cp_async_commit();
    }
    cp_async_wait<1>();
    __syncthreads();
    #pragma unroll
    for (int k = 0; k < BK; k++) {
      float hs[V3_TM];
      #pragma unroll
      for (int i = 0; i < V3_TM; i++) hs[i] = h_s[(mt * V3_TM + i) * H + kk * BK + k];
      const float4* bp0 = (const float4*)(cur + k * H + nt * 4);
      const float4* bp1 = (const float4*)(cur + k * H + 128 + nt * 4);
      const float4* bp2 = (const float4*)(cur + BK * H + k * H + nt * 4);
      const float4* bp3 = (const float4*)(cur + BK * H + k * H + 128 + nt * 4);
      float4 b0 = bp0[0], b1 = bp1[0], b2 = bp2[0], b3 = bp3[0];
      #pragma unroll
      for (int i = 0; i < V3_TM; i++) {
        float h = hs[i];
        azh[i][0] += h * b0.x; azh[i][1] += h * b0.y; azh[i][2] += h * b0.z; azh[i][3] += h * b0.w;
        azh[i][4] += h * b1.x; azh[i][5] += h * b1.y; azh[i][6] += h * b1.z; azh[i][7] += h * b1.w;
        azg[i][0] += h * b2.x; azg[i][1] += h * b2.y; azg[i][2] += h * b2.z; azg[i][3] += h * b2.w;
        azg[i][4] += h * b3.x; azg[i][5] += h * b3.y; azg[i][6] += h * b3.z; azg[i][7] += h * b3.w;
      }
    }
    __syncthreads();
  }
}

__global__ void __launch_bounds__(NTHREADS, 1) rollout_kernel_v3(
    const float* __restrict__ w_enc, const float* __restrict__ b_enc,
    const float* __restrict__ w_gru, const float* __restrict__ w_a, const float* __restrict__ b_a,
    float* __restrict__ agent0, float* __restrict__ food0, long long* __restrict__ rng0,
    float* __restrict__ state_buf, int* __restrict__ hit_arr,
    float* __restrict__ rewards, long long* __restrict__ positions,
    float* __restrict__ last_logits,
    int N, int horizon, unsigned long long* __restrict__ flag)
{
  extern __shared__ float smem[];
  float* h_s = smem;
  float* obs_s = h_s + V3_TILE * H;
  float* b_s = obs_s + V3_TILE * 4;

  const int tid = threadIdx.x;
  const int mt = tid >> 5;
  const int nt = tid & 31;
  int num_tiles = (N + V3_TILE - 1) / V3_TILE;

  for (int t = 0; t < horizon; t++) {
    for (int tile = blockIdx.x; tile < num_tiles; tile += gridDim.x) {
      int base = tile * V3_TILE;
      int nenv = N - base; if (nenv > V3_TILE) nenv = V3_TILE;

      if (tid < nenv) {
        int env = base + tid;
        float ax = agent0[env * 2], ay = agent0[env * 2 + 1];
        float fx = food0[env * 2], fy = food0[env * 2 + 1];
        obs_s[tid * 4 + 0] = (fx - ax) * (1.0f / (float)BOARD);
        obs_s[tid * 4 + 1] = (fy - ay) * (1.0f / (float)BOARD);
        obs_s[tid * 4 + 2] = ax * 0.1f;
        obs_s[tid * 4 + 3] = ay * 0.1f;
      }
      __syncthreads();

      for (int idx = tid; idx < nenv * H; idx += NTHREADS) {
        int e = idx / H, n = idx % H;
        const float* o = obs_s + e * 4;
        h_s[e * H + n] = o[0] * w_enc[n * 4 + 0] + o[1] * w_enc[n * 4 + 1]
                       + o[2] * w_enc[n * 4 + 2] + o[3] * w_enc[n * 4 + 3] + b_enc[n];
      }
      __syncthreads();

      for (int l = 0; l < L; l++) {
        const float* wl = w_gru + (size_t)l * G * H;
        float azh[V3_TM][V3_TN], azg[V3_TM][V3_TN];
        #pragma unroll
        for (int i = 0; i < V3_TM; i++)
          #pragma unroll
          for (int j = 0; j < V3_TN; j++) { azh[i][j] = 0.0f; azg[i][j] = 0.0f; }
        gemm_fused2_8(h_s, wl, wl + H * H, b_s, azh, azg, mt, nt, tid);
        #pragma unroll
        for (int i = 0; i < V3_TM; i++)
          #pragma unroll
          for (int j = 0; j < V3_TN; j++) {
            int e = mt * V3_TM + i, n = col_of(nt, j);
            if (e < nenv) {
              int env = base + e;
              float st = state_buf[(env * L + l) * H + n];
              float sg = sigf(azg[i][j]);
              float th = tanhf(azh[i][j]);
              float out = st + sg * (th - st);
              state_buf[(env * L + l) * H + n] = out;
              azh[i][j] = out;
            }
          }
        #pragma unroll
        for (int i = 0; i < V3_TM; i++)
          #pragma unroll
          for (int j = 0; j < V3_TN; j++) azg[i][j] = 0.0f;
        gemm_gate8(h_s, wl + 2 * H * H, b_s, azg, mt, nt, tid);
        __syncthreads();
        #pragma unroll
        for (int i = 0; i < V3_TM; i++)
          #pragma unroll
          for (int j = 0; j < V3_TN; j++) {
            int e = mt * V3_TM + i, n = col_of(nt, j);
            if (e < nenv) {
              float p = sigf(azg[i][j]);
              float out = azh[i][j];
              float ho = h_s[e * H + n];
              h_s[e * H + n] = p * out + (1.0f - p) * ho;
            }
          }
        __syncthreads();
      }

      logits_parallel(h_s, w_a, b_a, obs_s, tid, nenv);
      __syncthreads();
      if (tid < nenv) {
        int env = base + tid;
        float lg0 = obs_s[tid * 4 + 0], lg1 = obs_s[tid * 4 + 1];
        float lg2 = obs_s[tid * 4 + 2], lg3 = obs_s[tid * 4 + 3];
        int action = 0; float best = lg0;
        if (lg1 > best) { best = lg1; action = 1; }
        if (lg2 > best) { best = lg2; action = 2; }
        if (lg3 > best) { best = lg3; action = 3; }
        int dx = 0, dy = 0;
        if (action == 0) dy = -1;
        else if (action == 1) dy = 1;
        else if (action == 2) dx = -1;
        else dx = 1;
        float ax = agent0[env * 2], ay = agent0[env * 2 + 1];
        int nx = (int)ax + dx; if (nx < 0) nx = 0; if (nx > BOARD - 1) nx = BOARD - 1;
        int ny = (int)ay + dy; if (ny < 0) ny = 0; if (ny > BOARD - 1) ny = BOARD - 1;
        agent0[env * 2] = (float)nx; agent0[env * 2 + 1] = (float)ny;
        int hit = (nx == (int)food0[env * 2] && ny == (int)food0[env * 2 + 1]) ? 1 : 0;
        hit_arr[env] = hit;
        if (hit) { rewards[env] += 1.0f; atomicOr(&flag[t >> 6], 1ULL << (t & 63)); }
        if (t == horizon - 1) {
          last_logits[env * 4 + 0] = lg0;
          last_logits[env * 4 + 1] = lg1;
          last_logits[env * 4 + 2] = lg2;
          last_logits[env * 4 + 3] = lg3;
          positions[env * 2] = (long long)nx;
          positions[env * 2 + 1] = (long long)ny;
        }
      }
    }

    cg::this_grid().sync();
    int any_hit = (int)((atomicOr(&flag[t >> 6], 0) >> (t & 63)) & 1ULL);
    if (any_hit != 0) {
      for (int tile = blockIdx.x; tile < num_tiles; tile += gridDim.x) {
        int base = tile * V3_TILE;
        int nenv = N - base; if (nenv > V3_TILE) nenv = V3_TILE;
        for (int e = tid; e < nenv; e += NTHREADS) {
          int env = base + e;
          long long r = rng0[env];
          r = lcg_step(r);
          int nx2 = (int)(r % BOARD);
          r = lcg_step(r);
          int ny2 = (int)(r % BOARD);
          rng0[env] = r;
          if (hit_arr[env]) { food0[env * 2] = (float)nx2; food0[env * 2 + 1] = (float)ny2; }
        }
      }
    }
  }
}

// ---------------------------------------------------------------------------
// V4: TILE=32, TM=4, TN=8, fused zh+zg with single-buffer cp.async staging.
// Single-buffer halves the B shared footprint vs V3 (16KB vs 32KB) so 2 blocks
// of 256 threads fit per SM (16 warps, 33% occupancy) while keeping the async
// prefetch. The other resident block hides the exposed single-buffer latency.
// ---------------------------------------------------------------------------
#define V4_TILE 32
#define V4_TM 4
#define V4_TN 8

__device__ __forceinline__ void v4_gemm_fused2(const float* __restrict__ h_s,
                                               const float* __restrict__ Wzh,
                                               const float* __restrict__ Wzg,
                                               float* __restrict__ b_s,
                                               float azh[V4_TM][V4_TN], float azg[V4_TM][V4_TN],
                                               int mt, int nt, int tid) {
  const int nkt = H / BK;
  const int nq = BK * H / 4;
  const int nld = nq / NTHREADS;
  for (int kk = 0; kk < nkt; kk++) {
    const float4* Wf0 = (const float4*)(Wzh + kk * BK * H);
    const float4* Wf1 = (const float4*)(Wzg + kk * BK * H);
    float4* dst = (float4*)b_s;
    #pragma unroll
    for (int u = 0; u < nld; u++) {
      cp_async16((float*)(dst + u * NTHREADS + tid), (const float*)(Wf0 + u * NTHREADS + tid));
      cp_async16((float*)(dst + nq + u * NTHREADS + tid), (const float*)(Wf1 + u * NTHREADS + tid));
    }
    cp_async_commit();
    cp_async_wait<0>();
    __syncthreads();
    #pragma unroll
    for (int k = 0; k < BK; k++) {
      float hs[V4_TM];
      #pragma unroll
      for (int i = 0; i < V4_TM; i++) hs[i] = h_s[(mt * V4_TM + i) * H + kk * BK + k];
      const float4* bp0 = (const float4*)(b_s + k * H + nt * 4);
      const float4* bp1 = (const float4*)(b_s + k * H + 128 + nt * 4);
      const float4* bp2 = (const float4*)(b_s + BK * H + k * H + nt * 4);
      const float4* bp3 = (const float4*)(b_s + BK * H + k * H + 128 + nt * 4);
      float4 b0 = bp0[0], b1 = bp1[0], b2 = bp2[0], b3 = bp3[0];
      #pragma unroll
      for (int i = 0; i < V4_TM; i++) {
        float h = hs[i];
        azh[i][0] += h * b0.x; azh[i][1] += h * b0.y; azh[i][2] += h * b0.z; azh[i][3] += h * b0.w;
        azh[i][4] += h * b1.x; azh[i][5] += h * b1.y; azh[i][6] += h * b1.z; azh[i][7] += h * b1.w;
        azg[i][0] += h * b2.x; azg[i][1] += h * b2.y; azg[i][2] += h * b2.z; azg[i][3] += h * b2.w;
        azg[i][4] += h * b3.x; azg[i][5] += h * b3.y; azg[i][6] += h * b3.z; azg[i][7] += h * b3.w;
      }
    }
    __syncthreads();
  }
}

__device__ __forceinline__ void v4_gemm_gate(const float* __restrict__ h_s,
                                             const float* __restrict__ W,
                                             float* __restrict__ b_s,
                                             float acc[V4_TM][V4_TN],
                                             int mt, int nt, int tid) {
  const int nkt = H / BK;
  const int nq = BK * H / 4;
  const int nld = nq / NTHREADS;
  for (int kk = 0; kk < nkt; kk++) {
    const float4* Wf = (const float4*)(W + kk * BK * H);
    float4* dst = (float4*)b_s;
    #pragma unroll
    for (int u = 0; u < nld; u++)
      cp_async16((float*)(dst + u * NTHREADS + tid), (const float*)(Wf + u * NTHREADS + tid));
    cp_async_commit();
    cp_async_wait<0>();
    __syncthreads();
    #pragma unroll
    for (int k = 0; k < BK; k++) {
      float hs[V4_TM];
      #pragma unroll
      for (int i = 0; i < V4_TM; i++) hs[i] = h_s[(mt * V4_TM + i) * H + kk * BK + k];
      const float4* bp0 = (const float4*)(b_s + k * H + nt * 4);
      const float4* bp1 = (const float4*)(b_s + k * H + 128 + nt * 4);
      float4 b0 = bp0[0], b1 = bp1[0];
      #pragma unroll
      for (int i = 0; i < V4_TM; i++) {
        float h = hs[i];
        acc[i][0] += h * b0.x; acc[i][1] += h * b0.y; acc[i][2] += h * b0.z; acc[i][3] += h * b0.w;
        acc[i][4] += h * b1.x; acc[i][5] += h * b1.y; acc[i][6] += h * b1.z; acc[i][7] += h * b1.w;
      }
    }
    __syncthreads();
  }
}

      if (tid < nenv) {
        int env = base + tid;
        float lg0 = obs_s[tid * 4 + 0], lg1 = obs_s[tid * 4 + 1];
        float lg2 = obs_s[tid * 4 + 2], lg3 = obs_s[tid * 4 + 3];
        int action = 0; float best = lg0;
        if (lg1 > best) { best = lg1; action = 1; }
        if (lg2 > best) { best = lg2; action = 2; }
        if (lg3 > best) { best = lg3; action = 3; }
        int dx = 0, dy = 0;
        if (action == 0) dy = -1;
        else if (action == 1) dy = 1;
        else if (action == 2) dx = -1;
        else dx = 1;
        float ax = agent0[env * 2], ay = agent0[env * 2 + 1];
        int nx = (int)ax + dx; if (nx < 0) nx = 0; if (nx > BOARD - 1) nx = BOARD - 1;
        int ny = (int)ay + dy; if (ny < 0) ny = 0; if (ny > BOARD - 1) ny = BOARD - 1;
        agent0[env * 2] = (float)nx; agent0[env * 2 + 1] = (float)ny;
        int hit = (nx == (int)food0[env * 2] && ny == (int)food0[env * 2 + 1]) ? 1 : 0;
        hit_arr[env] = hit;
        if (hit) { rewards[env] += 1.0f; atomicOr(&flag[t >> 6], 1ULL << (t & 63)); }
        if (t == horizon - 1) {
          last_logits[env * 4 + 0] = lg0;
          last_logits[env * 4 + 1] = lg1;
          last_logits[env * 4 + 2] = lg2;
          last_logits[env * 4 + 3] = lg3;
          positions[env * 2] = (long long)nx;
          positions[env * 2 + 1] = (long long)ny;
        }
      }
    }

    cg::this_grid().sync();
    int any_hit = (int)((atomicOr(&flag[t >> 6], 0) >> (t & 63)) & 1ULL);
    if (any_hit != 0) {
      for (int tile = blockIdx.x; tile < num_tiles; tile += gridDim.x) {
        int base = tile * V4_TILE;
        int nenv = N - base; if (nenv > V4_TILE) nenv = V4_TILE;
        for (int e = tid; e < nenv; e += NTHREADS) {
          int env = base + e;
          long long r = rng0[env];
          r = lcg_step(r);
          int nx2 = (int)(r % BOARD);
          r = lcg_step(r);
          int ny2 = (int)(r % BOARD);
          rng0[env] = r;
          if (hit_arr[env]) { food0[env * 2] = (float)nx2; food0[env * 2 + 1] = (float)ny2; }
        }
      }
    }
  }
}

// ---------------------------------------------------------------------------
// V5: 128 threads/block, TM=8, TN=8, TILE=32, fused zh+zg with BK=4 double-
// buffered cp.async. 2 blocks/SM (2 x 128 x 255 regs fits; 2 x 48.5KB shared
// fits) so the other resident block hides barrier/latency stalls while keeping
// V3's 8-FMA-per-B-load arithmetic intensity.
// ---------------------------------------------------------------------------
#define V5_TILE 32
#define V5_TM 8
#define V5_TN 8
#define V5_BK 8
#define V5_NTHREADS 128

__device__ __forceinline__ void v5_gemm_fused2(const float* __restrict__ h_s,
                                               const float* __restrict__ Wzh, const float* __restrict__ Wzg,
                                               float* __restrict__ b_s,
                                               float azh[V5_TM][V5_TN], float azg[V5_TM][V5_TN],
                                               int mt, int nt, int tid) {
  float* buf0 = b_s;
  float* buf1 = b_s + 2 * V5_BK * H;
  const int nkt = H / V5_BK;
  const int nq = V5_BK * H / 4;
  const int nld = nq / V5_NTHREADS;
  {
    const float4* Wf0 = (const float4*)(Wzh);
    const float4* Wf1 = (const float4*)(Wzg);
    float4* dst = (float4*)buf0;
    #pragma unroll
    for (int u = 0; u < nld; u++) {
      cp_async16((float*)(dst + u * V5_NTHREADS + tid), (const float*)(Wf0 + u * V5_NTHREADS + tid));
      cp_async16((float*)(dst + nq + u * V5_NTHREADS + tid), (const float*)(Wf1 + u * V5_NTHREADS + tid));
    }
    cp_async_commit();
  }
  for (int kk = 0; kk < nkt; kk++) {
    float* cur = (kk & 1) ? buf1 : buf0;
    float* nxt = (kk & 1) ? buf0 : buf1;
    if (kk + 1 < nkt) {
      const float4* Wf0 = (const float4*)(Wzh + (kk + 1) * V5_BK * H);
      const float4* Wf1 = (const float4*)(Wzg + (kk + 1) * V5_BK * H);
      float4* dst = (float4*)nxt;
      #pragma unroll
      for (int u = 0; u < nld; u++) {
        cp_async16((float*)(dst + u * V5_NTHREADS + tid), (const float*)(Wf0 + u * V5_NTHREADS + tid));
        cp_async16((float*)(dst + nq + u * V5_NTHREADS + tid), (const float*)(Wf1 + u * V5_NTHREADS + tid));
      }
      cp_async_commit();
    }
    cp_async_wait<1>();
    __syncthreads();
    #pragma unroll
    for (int k = 0; k < V5_BK; k++) {
      float hs[V5_TM];
      #pragma unroll
      for (int i = 0; i < V5_TM; i++) hs[i] = h_s[(mt * V5_TM + i) * H + kk * V5_BK + k];
      const float4* bp0 = (const float4*)(cur + k * H + nt * 4);
      const float4* bp1 = (const float4*)(cur + k * H + 128 + nt * 4);
      const float4* bp2 = (const float4*)(cur + V5_BK * H + k * H + nt * 4);
      const float4* bp3 = (const float4*)(cur + V5_BK * H + k * H + 128 + nt * 4);
      float4 b0 = bp0[0], b1 = bp1[0], b2 = bp2[0], b3 = bp3[0];
      #pragma unroll
      for (int i = 0; i < V5_TM; i++) {
        float h = hs[i];
        azh[i][0] += h * b0.x; azh[i][1] += h * b0.y; azh[i][2] += h * b0.z; azh[i][3] += h * b0.w;
        azh[i][4] += h * b1.x; azh[i][5] += h * b1.y; azh[i][6] += h * b1.z; azh[i][7] += h * b1.w;
        azg[i][0] += h * b2.x; azg[i][1] += h * b2.y; azg[i][2] += h * b2.z; azg[i][3] += h * b2.w;
        azg[i][4] += h * b3.x; azg[i][5] += h * b3.y; azg[i][6] += h * b3.z; azg[i][7] += h * b3.w;
      }
    }
    __syncthreads();
  }
}

__device__ __forceinline__ void v5_gemm_gate(const float* __restrict__ h_s,
                                             const float* __restrict__ W,
                                             float* __restrict__ b_s,
                                             float acc[V5_TM][V5_TN],
                                             int mt, int nt, int tid) {
  float* buf0 = b_s;
  float* buf1 = b_s + V5_BK * H;
  const int nkt = H / V5_BK;
  const int nq = V5_BK * H / 4;
  const int nld = nq / V5_NTHREADS;
  {
    const float4* Wf = (const float4*)(W);
    float4* dst = (float4*)buf0;
    #pragma unroll
    for (int u = 0; u < nld; u++)
      cp_async16((float*)(dst + u * V5_NTHREADS + tid), (const float*)(Wf + u * V5_NTHREADS + tid));
    cp_async_commit();
  }
  for (int kk = 0; kk < nkt; kk++) {
    float* cur = (kk & 1) ? buf1 : buf0;
    float* nxt = (kk & 1) ? buf0 : buf1;
    if (kk + 1 < nkt) {
      const float4* Wf = (const float4*)(W + (kk + 1) * V5_BK * H);
      float4* dst = (float4*)nxt;
      #pragma unroll
      for (int u = 0; u < nld; u++)
        cp_async16((float*)(dst + u * V5_NTHREADS + tid), (const float*)(Wf + u * V5_NTHREADS + tid));
      cp_async_commit();
    }
    cp_async_wait<1>();
    __syncthreads();
    #pragma unroll
    for (int k = 0; k < V5_BK; k++) {
      float hs[V5_TM];
      #pragma unroll
      for (int i = 0; i < V5_TM; i++) hs[i] = h_s[(mt * V5_TM + i) * H + kk * V5_BK + k];
      const float4* bp0 = (const float4*)(cur + k * H + nt * 4);
      const float4* bp1 = (const float4*)(cur + k * H + 128 + nt * 4);
      float4 b0 = bp0[0], b1 = bp1[0];
      #pragma unroll
      for (int i = 0; i < V5_TM; i++) {
        float h = hs[i];
        acc[i][0] += h * b0.x; acc[i][1] += h * b0.y; acc[i][2] += h * b0.z; acc[i][3] += h * b0.w;
        acc[i][4] += h * b1.x; acc[i][5] += h * b1.y; acc[i][6] += h * b1.z; acc[i][7] += h * b1.w;
      }
    }
    __syncthreads();
  }
}

__global__ void __launch_bounds__(V5_NTHREADS, 1) rollout_kernel_v5(
    const float* __restrict__ w_enc, const float* __restrict__ b_enc,
    const float* __restrict__ w_gru, const float* __restrict__ w_a, const float* __restrict__ b_a,
    float* __restrict__ agent0, float* __restrict__ food0, long long* __restrict__ rng0,
    float* __restrict__ state_buf, int* __restrict__ hit_arr,
    float* __restrict__ rewards, long long* __restrict__ positions,
    float* __restrict__ last_logits,
    int N, int horizon, unsigned long long* __restrict__ flag)
{
  extern __shared__ float smem[];
  float* h_s = smem;
  float* obs_s = h_s + V5_TILE * H;
  float* b_s = obs_s + V5_TILE * 4;

  const int tid = threadIdx.x;
  const int mt = tid >> 5;
  const int nt = tid & 31;
  int num_tiles = (N + V5_TILE - 1) / V5_TILE;

  for (int t = 0; t < horizon; t++) {
    for (int tile = blockIdx.x; tile < num_tiles; tile += gridDim.x) {
      int base = tile * V5_TILE;
      int nenv = N - base; if (nenv > V5_TILE) nenv = V5_TILE;

      if (tid < nenv) {
        int env = base + tid;
        float ax = agent0[env * 2], ay = agent0[env * 2 + 1];
        float fx = food0[env * 2], fy = food0[env * 2 + 1];
        obs_s[tid * 4 + 0] = (fx - ax) * (1.0f / (float)BOARD);
        obs_s[tid * 4 + 1] = (fy - ay) * (1.0f / (float)BOARD);
        obs_s[tid * 4 + 2] = ax * 0.1f;
        obs_s[tid * 4 + 3] = ay * 0.1f;
      }
      __syncthreads();

      for (int idx = tid; idx < nenv * H; idx += V5_NTHREADS) {
        int e = idx / H, n = idx % H;
        const float* o = obs_s + e * 4;
        h_s[e * H + n] = o[0] * w_enc[n * 4 + 0] + o[1] * w_enc[n * 4 + 1]
                       + o[2] * w_enc[n * 4 + 2] + o[3] * w_enc[n * 4 + 3] + b_enc[n];
      }
      __syncthreads();

      for (int l = 0; l < L; l++) {
        const float* wl = w_gru + (size_t)l * G * H;
        float azh[V5_TM][V5_TN], azg[V5_TM][V5_TN];
        #pragma unroll
        for (int i = 0; i < V5_TM; i++)
          #pragma unroll
          for (int j = 0; j < V5_TN; j++) { azh[i][j] = 0.0f; azg[i][j] = 0.0f; }
        v5_gemm_fused2(h_s, wl, wl + H * H, b_s, azh, azg, mt, nt, tid);
        #pragma unroll
        for (int i = 0; i < V5_TM; i++)
          #pragma unroll
          for (int j = 0; j < V5_TN; j++) {
            int e = mt * V5_TM + i, n = col_of(nt, j);
            if (e < nenv) {
              int env = base + e;
              float st = state_buf[(env * L + l) * H + n];
              float sg = sigf(azg[i][j]);
              float th = tanhf(azh[i][j]);
              float out = st + sg * (th - st);
              state_buf[(env * L + l) * H + n] = out;
              azh[i][j] = out;
            }
          }
        #pragma unroll
        for (int i = 0; i < V5_TM; i++)
          #pragma unroll
          for (int j = 0; j < V5_TN; j++) azg[i][j] = 0.0f;
        v5_gemm_gate(h_s, wl + 2 * H * H, b_s, azg, mt, nt, tid);
        __syncthreads();
        #pragma unroll
        for (int i = 0; i < V5_TM; i++)
          #pragma unroll
          for (int j = 0; j < V5_TN; j++) {
            int e = mt * V5_TM + i, n = col_of(nt, j);
            if (e < nenv) {
              float p = sigf(azg[i][j]);
              float out = azh[i][j];
              float ho = h_s[e * H + n];
              h_s[e * H + n] = p * out + (1.0f - p) * ho;
            }
          }
        __syncthreads();
      }

      logits_parallel(h_s, w_a, b_a, obs_s, tid, nenv);
      __syncthreads();
      if (tid < nenv) {
        int env = base + tid;
        float lg0 = obs_s[tid * 4 + 0], lg1 = obs_s[tid * 4 + 1];
        float lg2 = obs_s[tid * 4 + 2], lg3 = obs_s[tid * 4 + 3];
        int action = 0; float best = lg0;
        if (lg1 > best) { best = lg1; action = 1; }
        if (lg2 > best) { best = lg2; action = 2; }
        if (lg3 > best) { best = lg3; action = 3; }
        int dx = 0, dy = 0;
        if (action == 0) dy = -1;
        else if (action == 1) dy = 1;
        else if (action == 2) dx = -1;
        else dx = 1;
        float ax = agent0[env * 2], ay = agent0[env * 2 + 1];
        int nx = (int)ax + dx; if (nx < 0) nx = 0; if (nx > BOARD - 1) nx = BOARD - 1;
        int ny = (int)ay + dy; if (ny < 0) ny = 0; if (ny > BOARD - 1) ny = BOARD - 1;
        agent0[env * 2] = (float)nx; agent0[env * 2 + 1] = (float)ny;
        int hit = (nx == (int)food0[env * 2] && ny == (int)food0[env * 2 + 1]) ? 1 : 0;
        hit_arr[env] = hit;
        if (hit) { rewards[env] += 1.0f; atomicOr(&flag[t >> 6], 1ULL << (t & 63)); }
        if (t == horizon - 1) {
          last_logits[env * 4 + 0] = lg0;
          last_logits[env * 4 + 1] = lg1;
          last_logits[env * 4 + 2] = lg2;
          last_logits[env * 4 + 3] = lg3;
          positions[env * 2] = (long long)nx;
          positions[env * 2 + 1] = (long long)ny;
        }
      }
    }

    cg::this_grid().sync();
    int any_hit = (int)((atomicOr(&flag[t >> 6], 0) >> (t & 63)) & 1ULL);
    if (any_hit != 0) {
      for (int tile = blockIdx.x; tile < num_tiles; tile += gridDim.x) {
        int base = tile * V5_TILE;
        int nenv = N - base; if (nenv > V5_TILE) nenv = V5_TILE;
        for (int e = tid; e < nenv; e += V5_NTHREADS) {
          int env = base + e;
          long long r = rng0[env];
          r = lcg_step(r);
          int nx2 = (int)(r % BOARD);
          r = lcg_step(r);
          int ny2 = (int)(r % BOARD);
          rng0[env] = r;
          if (hit_arr[env]) { food0[env * 2] = (float)nx2; food0[env * 2 + 1] = (float)ny2; }
        }
      }
    }
  }
}

def _mingru_g(x: torch.Tensor) -> torch.Tensor:
    return torch.tanh(x)


class Model(nn.Module):
    def __init__(self):
        super().__init__()
        self.w_enc = nn.Parameter(torch.empty(HIDDEN, OBS_DIM))
        self.b_enc = nn.Parameter(torch.zeros(HIDDEN))
        self.w_gru = nn.Parameter(torch.empty(GRU_LAYERS, GRU_OUT, HIDDEN))
        self.w_a = nn.Parameter(torch.empty(NUM_ACTIONS, HIDDEN))
        self.b_a = nn.Parameter(torch.zeros(NUM_ACTIONS))
        self.w_v = nn.Parameter(torch.empty(1, HIDDEN))
        self.b_v = nn.Parameter(torch.zeros(1))
        self.reset_parameters(0)

    def reset_parameters(self, seed: int = 0) -> None:
        g = torch.Generator(device="cpu")
        g.manual_seed(seed)
        for p in self.parameters():
            tmp = torch.empty(p.shape, dtype=p.dtype, device="cpu")
            tmp.normal_(0.0, 0.02, generator=g)
            p.data.copy_(tmp)

    def forward(self, obs: torch.Tensor, state: torch.Tensor):
        return policy_forward(self, obs, state)


def _w(model):
    w_enc = model.w_enc.detach().float().contiguous()
    b_enc = model.b_enc.detach().float().contiguous()
    w_gru = model.w_gru.detach().float().contiguous()
    w_a = model.w_a.detach().float().contiguous()
    b_a = model.b_a.detach().float().contiguous()
    w_v = model.w_v.detach().float().contiguous()
    b_v = model.b_v.detach().float().contiguous()
    return w_enc, b_enc, w_gru, w_a, b_a, w_v, b_v


def policy_forward(model: Model, obs: torch.Tensor, state: torch.Tensor):
    """obs (N,4), state (N,L,H) -> logits (N,4), new_state (N,L,H), value (N,)."""
    obs = obs.float().contiguous()
    state = state.float().contiguous()
    n = obs.shape[0]
    device = obs.device
    logits = torch.empty(n, NUM_ACTIONS, device=device, dtype=torch.float32)
    new_state = torch.empty(n, GRU_LAYERS, HIDDEN, device=device, dtype=torch.float32)
    value = torch.empty(n, device=device, dtype=torch.float32)
    w_enc, b_enc, w_gru, w_a, b_a, w_v, b_v = _w(model)
    _EXT.policy_launch(obs, state, w_enc, b_enc, w_gru, w_a, b_a, w_v, b_v,
                       logits, new_state, value, n)
    return logits, new_state, value


def _lcg_step(rng: torch.Tensor) -> torch.Tensor:
    return (rng * 6364136223846793005 + 1) & 0x7FFFFFFFFFFFFFFF


def env_step(agent, food, actions, rng_state):
    """Deterministic env step (matches reference exactly)."""
    delta = torch.zeros_like(agent)
    delta[:, 1] = torch.where(actions == 0, -torch.ones_like(delta[:, 1]), delta[:, 1])
    delta[:, 1] = torch.where(actions == 1, torch.ones_like(delta[:, 1]), delta[:, 1])
    delta[:, 0] = torch.where(actions == 2, -torch.ones_like(delta[:, 0]), delta[:, 0])
    delta[:, 0] = torch.where(actions == 3, torch.ones_like(delta[:, 0]), delta[:, 0])
    agent = (agent + delta).clamp(0, BOARD - 1)
    hit = (agent == food).all(dim=-1)
    reward = hit.float()
    rng_state = rng_state.clone()
    if hit.any():
        rng_state = _lcg_step(rng_state)
        fx = (rng_state % BOARD).to(agent.dtype)
        rng_state = _lcg_step(rng_state)
        fy = (rng_state % BOARD).to(agent.dtype)
        new_food = torch.stack([fx, fy], dim=-1)
        food = food.clone()
        food[hit] = new_food[hit]
    return agent, food, reward, rng_state


def run(num_envs: int, horizon: int, seed: int, model: Model | None = None) -> dict:
    device = torch.device("cuda:0" if torch.cuda.is_available() else "cpu")
    if model is None:
        model = Model()
    model = model.to(device).eval()

    # Identical initialization to the reference (CPU MT19937 randint).
    g = torch.Generator(device="cpu")
    g.manual_seed(seed)
    agent = torch.randint(0, BOARD, (num_envs, 2), generator=g).float().to(device)
    food = torch.randint(0, BOARD, (num_envs, 2), generator=g).float().to(device)
    rng_state = torch.arange(num_envs, device=device, dtype=torch.int64) + (seed * 10007)

    rewards = torch.zeros(num_envs, device=device, dtype=torch.float32)
    positions = torch.zeros(num_envs, 2, device=device, dtype=torch.int64)
    last_logits = torch.zeros(num_envs, NUM_ACTIONS, device=device, dtype=torch.float32)
    flag = torch.zeros(max(1, (horizon + 63) // 64), device=device, dtype=torch.int64)

    w_enc, b_enc, w_gru, w_a, b_a, w_v, b_v = _w(model)

    with torch.no_grad():
        if num_envs >= 1024:
            state_buf = torch.zeros(num_envs, GRU_LAYERS, HIDDEN, device=device, dtype=torch.float32)
            hit_arr = torch.zeros(num_envs, device=device, dtype=torch.int32)
            # V3 (TILE=64, TM=8/TN=8, fused zh+zg, 8 warps/SM) has the best
            # per-step GEMM throughput and wins when there are enough envs to
            # fill its 64-wide tiles, or enough horizon to amortize the tail.
            # V5 (TILE=32, 128-thread 1-block/SM) spreads its 32-wide tiles
            # across ~2x more SMs, so it wins for small/short runs where V3's
            # coarse tiles leave SMs idle.
            if num_envs >= 65536 or horizon >= 64:
                _EXT.rollout_launch_v3(w_enc, b_enc, w_gru, w_a, b_a,
                                       agent, food, rng_state, state_buf, hit_arr,
                                       rewards, positions, last_logits,
                                       num_envs, horizon, flag)
            else:
                _EXT.rollout_launch_v5(w_enc, b_enc, w_gru, w_a, b_a,
                                       agent, food, rng_state, state_buf, hit_arr,
                                       rewards, positions, last_logits,
                                       num_envs, horizon, flag)
        else:
            _EXT.rollout_launch(w_enc, b_enc, w_gru, w_a, b_a, w_v, b_v,
                                agent, food, rng_state, rewards, positions, last_logits,
                                num_envs, horizon, flag)

    return {
        "rewards": rewards,
        "positions": positions,
        "last_logits": last_logits,
    }
